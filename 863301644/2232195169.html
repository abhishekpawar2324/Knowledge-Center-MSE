Magic Global Support : Creation Of Symmetric/Assymetric Keys in MSSQL

Created by Amjad Madre on 31,May, 2023 
-- To delete
USE darpanDB;
DROP MASTER KEY;

--USE darpanDB;

-- to create certificate and keys

CREATE MASTER KEY ENCRYPTION BY PASSWORD = 'sasa@123';
CREATE CERTIFICATE MagicDBBusinessAlfaCertif WITH SUBJECT = 'Certificate for Sym_Key_Conns';
CREATE SYMMETRIC KEY Sym_Key_Conns WITH ALGORITHM = AES_256 ENCRYPTION BY CERTIFICATE MagicDBBusinessAlfaCertif;
-- Table Creation
CREATE TABLE Contacts1 (
    ID INT PRIMARY KEY,
    First_Name VARCHAR(50),
    Encrypted_Last_Name VARBINARY(100),
    Date_Created DATETIME
)
-- Insert the Encrypted Data
OPEN SYMMETRIC KEY Sym_Key_Conns DECRYPTION BY CERTIFICATE MagicDBBusinessAlfaCertif;
INSERT INTO Contacts1 VALUES (1, 'Darpan', ENCRYPTBYKEY(KEY_GUID('Sym_Key_Conns'), 'Bandil'), GETDATE());
CLOSE SYMMETRIC KEY Sym_Key_Conns;
- Execute Below
OPEN SYMMETRIC KEY Sym_Key_Conns DECRYPTION BY CERTIFICATE MagicDBBusinessAlfaCertif;
SELECT ID, First_Name, CONVERT(VARCHAR(100), DECRYPTBYKEY(Encrypted_Last_Name)) AS Last_Name, Date_Created FROM Contacts1;
CLOSE SYMMETRIC KEY Sym_Key_Conns;
