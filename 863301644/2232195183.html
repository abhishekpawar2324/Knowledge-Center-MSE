               [First_Name] [varchar](50) NULL,

               [Last_Name] [varbinary](max) NULL,

) ON [PRIMARY] TEXTIMAGE_ON [PRIMARY]

Insert statement:

insert into Contacts1 values (1,'Darpan',CONVERT(varbinary(32), 'Bandil'),GETDATE())

update statement:

--UPDATE Contacts1 SET Last_Name = ENCRYPTBYKEY(KEY_GUID('skey'), Last_Name)

Keys to create

CREATE ASYMMETRIC KEY akey WITH ALGORITHM = RSA_2048 ENCRYPTION BY PASSWORD = 'aaa123@'

GO

CREATE SYMMETRIC KEY skey WITH ALGORITHM = AES_256 ENCRYPTION BY ASYMMETRIC KEY akey

GO

 

To execute as 3 statements:
OPEN SYMMETRIC KEY skey DECRYPTION BY ASYMMETRIC KEY akey WITH PASSWORD = 'aaa123@'

select CAST(DECRYPTBYKEY(Last_Name) as VARCHAR(100)) from contacts1

CLOSE SYMMETRIC KEY skey

Solution: Create a procedure combining three statements:

ALTER proc [dbo].[symkeyproc2]

    WITH EXECUTE AS OWNER

 

as

begin

    OPEN SYMMETRIC KEY skey DECRYPTION BY ASYMMETRIC KEY akey WITH PASSWORD = 'aaa123@';

select CAST(DECRYPTBYKEY(Last_Name) as VARCHAR(100)) from contacts1;

CLOSE SYMMETRIC KEY skey;

end

So the user has permission to execute only permission of that Stored Procedure.
